# Configuración de arranque recomendada para MeteoSer
# Guarda este archivo como 'meteoser_configuracion.txt' o similar para referencia rápida

# Arrancar el servidor accesible desde toda la red local:
# (Sustituye 192.168.x.x por tu IP local real)
uvicorn main_asgi:app --host 192.168.x.x --port 8080 --reload

# Acceso desde el navegador:
# http://192.168.x.x:8080

# Si solo lo quieres en tu PC:
uvicorn main_asgi:app --host 127.0.0.1 --port 8080 --reload

# O usando el script:
python arrancar_meteoser.py

# Si necesitas cambiar el puerto, solo modifica el número tras --port
# Ejemplo: --port 8090

# Para ver tu IP local ejecuta en PowerShell:
ipconfig
