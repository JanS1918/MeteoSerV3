import numba as nb
import numpy as np

@nb.jit(nopython=True)  # Compilar a C automáticamente
def calcular_densidad_vectorizado(T_array, P_array, HR_array):
    densidades = np.zeros(len(T_array))
    for i in range(len(T_array)):
        # Fórmula de densidad
        T_k = T_array[i] + 273.15
        e = HR_array[i] / 100 * 6.1094 * np.exp(17.625 * T_array[i] / (T_array[i] + 243.04))
        densidades[i] = (P_array[i] - 0.3783 * e) / (287.05 * T_k)
    return densidades

# Usar
T = np.array([20, 21, 22, 23, 24, 25])  # 1000 temperaturas
P = np.array([1013]*6)
HR = np.array([60]*6)
densidades = calcular_densidad_vectorizado(T, P, HR)  # 100x más rápido

from sklearn.linear_model import LinearRegression

# Recolectar datos: sensor vs "verdad" (referencia confiable)
X_sensor = np.array([20.1, 20.3, 20.5, 20.7, 21.0])
y_verdad = np.array([20.0, 20.0, 20.0, 20.0, 20.0])

# Entrenar corrección
modelo = LinearRegression()
modelo.fit(X_sensor.reshape(-1, 1), y_verdad)

# Auto-aplicar
def temperatura_corregida(temp_sensor):
    return modelo.predict([[temp_sensor]])[0]

# En Bus
T_raw = bus.capa_core.obtener_valor("sensores.temperatura")
T_corregida = temperatura_corregida(T_raw)
bus.publicar("sensores.temperatura_calibrada", T_corregida, "CORE")

from influxdb_client import InfluxDBClient

client = InfluxDBClient(url="http://localhost:8086", token="mi_token", org="mi_org")
write_api = client.write_api(write_options=SYNCHRONOUS)

def guardar_al_bus():
    """Corre cada minuto, guarda variables del Bus"""
    core_vars = bus.query("core:*")
    
    for meta in core_vars:
        punto = {
            "measurement": "meteoser",
            "tags": {"variable": meta.variable},
            "fields": {"valor": meta.valor, "confianza": meta.confianza},
            "time": meta.timestamp
        }
        write_api.write(bucket="meteoser", record=punto)
