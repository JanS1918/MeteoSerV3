  def calcular_radiacion_final(self, sensores_dict, rest2_output):
      """
      Nueva versión: usa wrapper robusto.
      """
      from core.radiation.wrapper_integracion import WrapperRadiacionRobusta
      
      # Crear wrapper si no existe
      if not hasattr(self, '_wrapper_radiacion'):
          self._wrapper_radiacion = WrapperRadiacionRobusta(
              enabled=True,
              modo_comparativa=False  # Ya no comparativa, ahora productivo
          )
      
      # Procesar con wrapper (ahora USAR resultado)
      resultado = self._wrapper_radiacion.procesar(
          elevacion_solar_deg=sensores_dict.get('elevacion_solar', 0),
          ghi_w_m2_medida=sensores_dict.get('ghi_sensor'),
          presion_hpa=sensores_dict.get('presion_hpa'),
          humedad_rel=sensores_dict.get('humedad_rel'),
          temperatura_c=sensores_dict.get('temperatura_c'),
          velocidad_viento_ms=sensores_dict.get('velocidad_viento',0),
          precipitacion_mm=sensores_dict.get('precipitacion', 0),
          visibilidad_km=sensores_dict.get('visibilidad_km'),
          rest2_output=rest2_output,
          sensor_real_confiable=True
      )
      
      # Retornar resultado final (ahora de arquitectura robusta)
      return {
          'ghi': resultado['ghi'],
          'dni': resultado['dni'],
          'dhi': resultado['dhi'],
          'confianza': resultado['confianza'],
          'contexto': resultado['contexto'],
          'advertencias': resultado['advertencias']
      }
