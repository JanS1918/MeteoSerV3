💡 Uso Rápido:
   from core.persistence import InfluxDBManager, obtener_archivador
   from datetime import datetime, timedelta
   
   # 1. Conectar a BD
   db = InfluxDBManager(
       url="http://localhost:8086",
       token="tu-token",
       bucket="meteodata"
   )
   db.conectar()
   
   # 2. Iniciar archivado automático
   archivador = obtener_archivador(bus, db)
   archivador.iniciar(intervalo_segundos=30)
   
   # El sistema guarda automáticamente todo el Bus cada 30s
   
   # 3. Consultar histórico
   datos = archivador.recuperar_historico(
       "sensores.temperatura_c",
       inicio=datetime.now() - timedelta(days=7),
       agregacion="mean",
       ventana="5m"
   )
   
   # 4. Estadísticas
   stats = archivador.obtener_estadisticas_variable("temperatura", dias=7)
   # → {min: 15.2, max: 32.1, mean: 22.5, count: 10080}
   
   # 5. Exportar CSV
   csv_path = archivador.exportar_a_csv(
       "temperatura",
       inicio=datetime.now() - timedelta(days=7)
   )
   
   # 6. Al cerrar
   archivador.detener()
   db.desconectar()
