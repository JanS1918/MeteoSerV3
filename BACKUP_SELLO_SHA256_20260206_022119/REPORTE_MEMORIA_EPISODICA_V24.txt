```python
class AIAssistant:
    def __init__(self, llm_client=None, memory=None):
        self.llm_client = llm_client
        self.memory = memory  # <-- NUEVA
    
    async def explain_anomaly(self, detection_result, context):
        # Buscar anomalias similares pasadas
        if self.memory:
            similar = self.memory.find_similar_anomalies(detection_result)
            if similar:
                # Aprender de resoluciones previas
                past_resolution = similar[0]["resolution_type"]
                explanation += f" (Similar caso pasado resuelto con: {past_resolution})"
        
        return explanation
```
